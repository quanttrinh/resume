% --- Body typeface: EB Garamond -------------------------------------------
% On a Unicode engine (XeTeX/LuaTeX -- what tectonic runs) the OTFs are loaded
% directly. This avoids ebgaramond.sty, which switches on XeTeX font tracing
% and probes EBGaramond-Initials, producing pages of harmless build warnings.
% pdfLaTeX has no fontspec, so it falls back to the package's Type 1 route.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{ebgaramond}
\else
  \usepackage{fontspec}
  \setmainfont{EBGaramond}[
    Extension      = .otf,
    UprightFont    = *-Regular,
    ItalicFont     = *-Italic,
    BoldFont       = *-Bold,
    BoldItalicFont = *-BoldItalic,
    Ligatures      = TeX,
    Numbers        = {Proportional,OldStyle},
  ]
\fi
