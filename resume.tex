% --- Body typeface: XCharter ------------------------------------------------
% A Charter derivative: large x-height, sturdy at small sizes, and it holds up
% on screen where an old-style face like Garamond reads thin.
% On a Unicode engine (XeTeX/LuaTeX -- what tectonic runs) the OTFs are loaded
% directly; pdfLaTeX has no fontspec, so it falls back to the package.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{XCharter}
\else
  \usepackage{fontspec}
  \setmainfont{XCharter}[
    Extension      = .otf,
    UprightFont    = *-Roman,
    ItalicFont     = *-Italic,
    BoldFont       = *-Bold,
    BoldItalicFont = *-BoldItalic,
    Ligatures      = TeX,
  ]
\fi
